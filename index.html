    xaxis_title="AGE",

)

import plotly.graph_objects as go

stats = ["PTS", "AST", "TRB", "STL", "BLK", "WS", "eFG%_norm", "team_win_pct"]
metrics = {"mvp_score": "Weighted Formula Score", "mvp_probability": "ML Model Probability"}

fig = go.Figure()
visible_index = 0
buttons = []
trace_index = 0

for stat in stats: 
    for metric_col, metric_label in metrics.items():
        fig.add_trace(go.Scatter(
            x = players_df[stat], y = players_df[metric_col],
            mode='markers', text=players_df['Player'],
            name= f"{stat} vs {metric_label}",
            visible = (trace_index == 0)
        ))
        trace_index += 1
        
trace_index = 0 
for stat in stats: 
    for metric_col, metric_label in metrics.items():
        visibility = [i == trace_index for i in range(len(stats) * len(metrics))]
        buttons.append(dict(
            label = f"{stat} vs {metric_label}",
            method = "update",
            args = [{"visible": visibility}, {"title": f"{stat} vs {metric_label}", "xaxis.title.text": f"{stat}", "yaxis.title.text": f"{metric_label}"}]
        ))
        trace_index += 1
fig.update_layout(
    updatemenus=[dict(active=0, buttons=buttons, x=0.5, y=1.15)],
    #title=f"{stats[0]} vs {list(metrics.values())[0]}",
    xaxis_title=stats[0],
    yaxis_title=list(metrics.values())[0],
    template="plotly_dark"

)
fig.update_layout(margin=dict(t=100))
fig.show()
